cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(CurveController VERSION 1.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use Visual Studio 2022 on Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
add_compile_options(/W4 /WX /utf-8 /permissive- /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /PDBALTPATH:%_PDB%)
option(CURVE_ASAN "AddressSanitizer validation; not for distribution" OFF)
if(CURVE_ASAN)
  add_compile_options(/fsanitize=address /Zi)
  add_link_options(/DEBUG /INCREMENTAL:NO)
endif()
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
  set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/lib")
endforeach()
add_library(InterceptorSDK STATIC Interceptor/Interceptor.cpp)
target_include_directories(InterceptorSDK PUBLIC Interceptor)
target_link_libraries(InterceptorSDK PUBLIC setupapi newdev cfgmgr32 hid advapi32 user32 comctl32 ole32)
add_library(CurveCore STATIC App/Curve.cpp)
target_include_directories(CurveCore PUBLIC App)
add_library(CurveRuntime STATIC App/Transform.cpp App/Storage.cpp App/Capture.cpp)
target_link_libraries(CurveRuntime PUBLIC CurveCore InterceptorSDK)
add_library(CurveUI STATIC App/UI.cpp)
add_library(DriverSetup STATIC App/DriverSetup.cpp)
target_link_libraries(DriverSetup PUBLIC InterceptorSDK bcrypt shell32 comctl32)
target_link_libraries(CurveUI PUBLIC CurveRuntime DriverSetup d2d1 dwrite windowscodecs shell32 ole32 uuid version)
add_executable(CurveController WIN32 App/Main.cpp App/App.rc)
target_link_libraries(CurveController PRIVATE CurveUI)
target_link_options(CurveController PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/App/App.manifest")
enable_testing()
add_executable(CurveTests App/Tests/Tests.cpp)
target_link_libraries(CurveTests PRIVATE CurveRuntime)
add_test(NAME CurveAndControllerSemantics COMMAND CurveTests "${CMAKE_CURRENT_SOURCE_DIR}/Examples")
